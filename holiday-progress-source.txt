function holidayStorySchedule(holiday){
  const standard=[
    {
      key:"story-1",
      label:"2 dias úteis antes",
      date:holidayBusinessDayBefore(holiday.date,2)
    },
    {
      key:"story-2",
      label:"1 dia útil antes",
      date:holidayBusinessDayBefore(holiday.date,1)
    },
    {
      key:"story-3",
      label:"No feriado",
      date:holiday.date
    }
  ];

  const today=isoDate(new Date());
  if(today<standard[0].date)return standard;
  if(today>holiday.date)return[];

  const remaining=standard.filter(entry=>entry.date>today);
  return[
    {
      key:"story-now",
      label:"Publicar imediatamente",
      date:today
    },
    ...remaining
  ].filter((entry,index,list)=>
    list.findIndex(candidate=>candidate.date===entry.date)===index
  );
}

function holidayOperationSentence(holiday,value){
  const date=holidayLongDate(holiday.date);
  const weekdayName=holidayWeekday(holiday.date);

  if(value.expediente==="Fechado"){
    return`No feriado de ${holiday.name}, em ${date}, ${weekdayName}, estaremos fechados.`;
  }
  if(value.expediente==="Somente acolhimento fechado"){
    return`No feriado de ${holiday.name}, em ${date}, ${weekdayName}, não haverá acolhimento.`;
  }
  if(value.expediente==="Somente expedição fechada"){
    return`No feriado de ${holiday.name}, em ${date}, ${weekdayName}, a expedição estará pausada.`;
  }
  if(value.expediente==="Horário reduzido"){
    const schedule=(value.horario_escala||"").trim()||"[CONFIRMAR HORÁRIO REDUZIDO]";
    return`No feriado de ${holiday.name}, em ${date}, ${weekdayName}, estaremos online em horário especial: ${schedule}.`;
  }
  if(value.expediente==="Plantão"){
    const schedule=(value.horario_escala||"").trim()||"[CONFIRMAR HORÁRIO OU ESCALA DE PLANTÃO]";
    return`No feriado de ${holiday.name}, em ${date}, ${weekdayName}, teremos plantão: ${schedule}.`;
  }
  if(value.expediente==="Abertura excepcional"){
    return`No feriado de ${holiday.name}, em ${date}, ${weekdayName}, vamos estar por aqui.`;
  }
  return`O funcionamento da Adapta em ${date} precisa ser confirmado.`;
}

        item.natures=["Pauta","Informação","Feriado"];
        item.priority="normal";
        item.behavior="main";
        item.formats="Feed · formato a definir";
        item.stageChangedAt=nowIso();

      }

      state.items[id]=item;

      generated.push({
        id,
        number:201+index,
        code:`PF${String(index+1).padStart(2,"0")}`,
        title:`Pauta de feed · ${holiday.name}`,
        base:"Informativos feriados",
        tags:[
          "Informativos feriados",
          "Pauta editorial",
          "Feed",
          holiday.name,
          "Gaby"
        ],
        bodyHtml:holidayFeedBriefingHtml(holiday,value),
        source:"holiday-feed",
        holidayId:holiday.id,
        active:true,
        generatedAt:nowIso()
      });
    }
  });

  state.holidayContents=generated;
}

function centralHolidayTaskStatus(holiday,value){
  if(value.expediente==="A definir"){
    return{status:"bloqueado",label:"Definir expediente"};
  }
  if(value.feed_editorial==="A decidir"){
    return{status:"bloqueado",label:"Definir pauta de feed"};
  }
  if(holidayIsNoChange(value)){
    return value.feed_editorial==="Relevante"
      ?{status:"concluido",label:"Pauta de feed criada"}
      :{status:"dispensado",label:"Sem aviso de feriado"};
  }

  const teamOk=["Equipe avisada","Não necessário"].includes(value.equipe_interna)
    ||(value.equipe_acolhimento&&value.equipe_expedicao);

  if(!teamOk){
    return{status:"bloqueado",label:"Alinhar equipe interna"};
  }
  if(!["Enviado","Não necessário"].includes(value.email_associados)){
    return{status:"pendente",label:"Preparar ou enviar e-mail"};
  }
  if(value.instagram!=="Agendado"){
    return{status:"pendente",label:"Agendar os Stories"};
  }
  return{status:"concluido",label:"Comunicação organizada"};
}

function syncHolidayRecordsToCentral(){
  const tasks=[];
  HOLIDAYS_2026.forEach(holiday=>{
    const value=centralHolidayState(holiday);
    const communication=centralHolidayTaskStatus(holiday,value);
    const entry=state.calendarEntries.find(item=>item.id===`feriado-${holiday.date}`);
    const complete=centralHolidayComplete(value);

    if(entry){
      const normalOperation=holidayIsNoChange(value);
      entry.title=normalOperation
        ?`Feriado — ${holiday.name} · sem alteração`
        :`Informativo — ${holiday.name}`;
      entry.operationDecision=value.expediente;
      entry.behavior=normalOperation?"parallel":"main";
      entry.status=complete?"confirmado":"provisório";
      entry.note=[
        `Expediente: ${value.expediente}`,
        value.horario_escala?`Horário/Escala: ${value.horario_escala}`:"",
        `Equipe: ${value.equipe_interna}`,
        `Acolhimento: ${value.equipe_acolhimento?"confirmado":"pendente"}`,
        `Expedição: ${value.equipe_expedicao?"confirmada":"pendente"}`,
        `E-mail: ${value.email_associados}`,
        `Stories: ${value.instagram}`,
        `Pauta de feed: ${value.feed_editorial}`,
        value.observacao?`Observação: ${value.observacao}`:""
      ].filter(Boolean).join(" · ");
      entry.updatedAt=value.updated_at||nowIso();
    }

    tasks.push({
      holidayId:holiday.id,
      title:holiday.name,
      date:holiday.date,
      scope:holiday.scope,
      status:communication.status,
      label:communication.label,
      complete,
      expediente:value.expediente,
      email:value.email_associados,
      instagram:value.instagram
    });
  });
  state.holidayCommunicationTasks=tasks;
  syncHolidayContentBase();
}

function renderHolidayOverview(){
  const container=document.getElementById("holidayOverviewList");
  const count=document.getElementById("holidayOverviewCount");
  if(!container||!count)return;

  const todayDate=new Date();
  todayDate.setHours(0,0,0,0);
  const upcoming=(state.holidayCommunicationTasks||[])
    .filter(task=>dateFromIso(task.date)>=todayDate)
    .sort((a,b)=>a.date.localeCompare(b.date));

  const pending=upcoming.filter(task=>!["concluido","dispensado"].includes(task.status));
  count.textContent=`${pending.length} pendências`;

  const display=(pending.length?pending:upcoming).slice(0,5);
  container.innerHTML=display.length?display.map(task=>`
    <div class="holiday-overview-row">
      <div>
        <strong>${escapeHtml(task.title)}</strong>
        <span>${formatDate(task.date)} · ${escapeHtml(task.expediente)}</span>
      </div>
      <button class="link-button" type="button" data-open-holiday-page>
        ${escapeHtml(task.label)}
      </button>
    </div>
  `).join(""):`<div class="empty">Não há feriados futuros neste calendário.</div>`;

  container.querySelectorAll("[data-open-holiday-page]").forEach(button=>{
    button.addEventListener("click",()=>openPage("holidays"));
  });
}

function hydrateHolidayRecords(){
    if(hydrating) return;
    hydrating=true;
    try{
      const response=await fetch("/api/holiday-records",{cache:"no-store",credentials:"same-origin"});
      if(!response.ok) return;
      const result=await response.json();
      if(!Array.isArray(result.holidayRecords)) return;
      state.holidayRecords=JSON.parse(JSON.stringify(result.holidayRecords));
      if(Array.isArray(result.holidayCommunicationTasks)) state.holidayCommunicationTasks=JSON.parse(JSON.stringify(result.holidayCommunicationTasks));
      if(Array.isArray(result.holidayContents)) state.holidayContents=JSON.parse(JSON.stringify(result.holidayContents));
      try{localStorage.setItem(KEY,JSON.stringify(state));}catch{}
      try{refreshWithoutCalendarRebuild();}catch{}
      try{holidayDataHandler?.onDataChanged(JSON.parse(JSON.stringify(state.holidayRecords)));}catch{}
    }catch(error){ console.warn("Não foi possível sincronizar feriados do D1.",error); }
    finally{hydrating=false;}
  }
  if(document.readyState==="loading") document.addEventListener("DOMContentLoaded",hydrateHolidayRecords,{once:true});
  else hydrateHolidayRecords();
  document.addEventListener("click",event=>{
    if(event.target?.closest?.('[data-page="holidays"]')) setTimeout(hydrateHolidayRecords,30);
  });
}
