    async function persistHolidayRecord(record){
      const declaredBy = String(
        record.log_signer
        || sessionStorage.getItem("central_adapta_declared_by")
        || ""
      ).trim();
      if(!declaredBy) return {isOk:false,error:"declared_by_required"};

      try{
        const response = await fetch("/api/holiday-record",{
          method:"PUT",
          headers:{"content-type":"application/json"},
          credentials:"same-origin",
          body:JSON.stringify({record,declaredBy})
        });
        const result = await response.json().catch(()=>({}));
        if(!response.ok) return {isOk:false,error:result.error||"save_failed",result};

        sessionStorage.setItem("central_adapta_declared_by",declaredBy);
        state.holidayRecords = Array.isArray(result.holidayRecords)
          ? cloneHolidayData(result.holidayRecords)
          : state.holidayRecords;
        localStorage.setItem(STORAGE_KEY,JSON.stringify(state));
        refreshWithoutCalendarRebuild();
        holidayDataHandler?.onDataChanged(cloneHolidayData(state.holidayRecords));
        return {isOk:true,revision:result.revision,loggedChanges:result.loggedChanges};
      }catch(error){
        console.error("Falha ao persistir feriado no D1",error);
        return {isOk:false,error:"network_error"};
      }
    }

          displayText = "Equipe avisada";
          displayType = "done";
        }
      }
      
      chip.className = "status-chip " + displayType;
      chip.textContent = displayText;
      const icon = document.createElement("i");
      icon.setAttribute("data-lucide","chevron-down");
      button.append(chip,icon);
      
      button.addEventListener("click",event => {
        event.stopPropagation();
        openFieldEditor(id,field,button);
      });
      return button;
    }

    function makeInstagramAction(id,value){
      const holiday = holidayById(id);
      const isWk = holiday && isWeekend(holiday.date);
      
      const checked = value === "Agendado";
      const button = document.createElement("button");
      button.type = "button";
      button.className = "instagram-action" + (checked ? " is-checked" : "");
      if (isWk) {
        button.disabled = true;
        button.style.opacity = "0.5";
        button.style.cursor = "not-allowed";
      }
      button.setAttribute("aria-pressed",String(checked));
      button.setAttribute("aria-label",checked ? "Desmarcar os Stories como agendados" : "Marcar os três Stories como agendados");
      const box = document.createElement("span");
      box.className = "instagram-box";
      const icon = document.createElement("i");
      icon.setAttribute("data-lucide","check");
      box.appendChild(icon);
      const label = document.createElement("span");
      label.textContent = checked ? "Agendado" : "Agendar";
      button.append(box,label);
      
      button.addEventListener("click",event => {
        event.stopPropagation();
        closeFieldEditor();
        savePatch(id,{instagram:checked ? "Não agendado" : "Agendado"},"Stories atualizados.");
      });
      return button;
    }

    function holidayCountdownLabel(value){
      const target = dateObject(value);
      target.setHours(0,0,0,0);

      const reference = new Date(today);
      reference.setHours(0,0,0,0);

      const difference = Math.round(
        (target.getTime() - reference.getTime()) / (1000 * 60 * 60 * 24)
      );

      if(difference === 0) return "hoje";
      if(difference === 1) return "falta 1 dia";
      if(difference > 1) return `faltam ${difference} dias`;
      if(difference === -1) return "foi ontem";
      return `há ${Math.abs(difference)} dias`;
    }
